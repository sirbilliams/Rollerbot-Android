find_package(SDL2 REQUIRED)
if(NOT TARGET rb_assets)
  add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../src/assets ${CMAKE_CURRENT_BINARY_DIR}/assets)
endif()
add_executable(rbview rbview.cpp)
target_link_libraries(rbview PRIVATE rb_scene rb_assets SDL2::SDL2)
if(BUILD_TESTING)
  add_executable(rb_scene_test ${CMAKE_CURRENT_LIST_DIR}/../../tests/scene/scene_test.cpp)
  target_link_libraries(rb_scene_test PRIVATE rb_scene rb_assets SDL2::SDL2)
  add_test(NAME rb_scene COMMAND rb_scene_test)
endif()
if(BUILD_TESTING)
  # Xvfb has no GLX ES profile: force EGL. Run as: xvfb-run -a ctest
  set_tests_properties(rb_scene PROPERTIES ENVIRONMENT "SDL_VIDEO_X11_FORCE_EGL=1" SKIP_RETURN_CODE 77)
endif()
