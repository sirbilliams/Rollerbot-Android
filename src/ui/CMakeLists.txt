cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rb_ui_lane LANGUAGES CXX)
  enable_testing()
endif()
add_library(rb_ui Ui.cpp)
target_compile_features(rb_ui PUBLIC cxx_std_17)
target_include_directories(rb_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
option(RB_UI_TESTS "Build UI lane tests" ON)
if(RB_UI_TESTS)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../tests/ui ${CMAKE_CURRENT_BINARY_DIR}/tests)
endif()
