cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(rb_audio_standalone LANGUAGES CXX)
 enable_testing()
endif()
find_package(SDL2 REQUIRED)
add_library(rb_audio Audio.cpp Movie.cpp)
target_compile_features(rb_audio PUBLIC cxx_std_17)
target_include_directories(rb_audio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_link_libraries(rb_audio PUBLIC SDL2::SDL2)
option(RB_AUDIO_TESTS "Build audio fixtures" ON)
if(RB_AUDIO_TESTS)
 add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../tests/audio ${CMAKE_CURRENT_BINARY_DIR}/tests)
endif()
