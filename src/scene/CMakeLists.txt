cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rb_scene_standalone LANGUAGES CXX)
  enable_testing()
endif()
if(NOT RB_ANDROID)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GLES2 REQUIRED IMPORTED_TARGET glesv2)
endif()
add_library(rb_scene Scene.cpp)
target_compile_features(rb_scene PUBLIC cxx_std_17)
target_include_directories(rb_scene PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
if(RB_ANDROID)
  # The NDK has no pkg-config; the top level found libGLESv2 directly.
  target_link_libraries(rb_scene PUBLIC ${RB_GLES2})
else()
  target_link_libraries(rb_scene PUBLIC PkgConfig::GLES2)
endif()
option(RB_SCENE_TOOLS "Build rbview and scene tests" ON)
if(RB_SCENE_TOOLS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/render/CMakeLists.txt)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../tools/render ${CMAKE_CURRENT_BINARY_DIR}/rbview)
endif()
