# Android native build: SDL2 (shared) + the repo's top-level CMake (libs: rollerbot)
cmake_minimum_required(VERSION 3.16)
project(rollerbot_app C CXX)
if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/SDL/CMakeLists.txt)
  message(FATAL_ERROR "SDL sources are missing - run ./fetch-deps.sh from the repo root first")
endif()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(SDL)
set(RB_ANDROID ON CACHE BOOL "" FORCE)
set(RB_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../.. rollerbot)
